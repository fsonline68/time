    <div class="card">

        <div id="status">Bereit</div>

        <div class="controls">

            <button class="btn-start" id="startBtn" onclick="clockIn()">Start</button>

            <button class="btn-pause" id="pauseBtn" onclick="togglePause()" disabled>Pause</button>

            <button class="btn-stop" id="stopBtn" onclick="clockOut()" disabled>Stopp</button>

            <button class="btn-csv" onclick="exportCSV()">CSV Export</button>

        </div>

    <div class="card stats" id="statsContainer">

        <h3>Fortschreibung</h3>

        <div id="statsContent">Lade Daten...</div>

    </div>


    <script>

        let entries = JSON.parse(localStorage.getItem('time_entries') |


| '');

        let timerInterval;


        function save() {

            localStorage.setItem('time_entries', JSON.stringify(entries));

            renderStats();

        }


        function clockIn() {

            const now = Date.now();

            entries.push({ type: 'WORK', start: now, end: null });

            updateUI('WORKING');

            save();

        }


        function togglePause() {

            const now = Date.now();

            const last = entries[entries.length - 1];

            if (last.end === null) {

                last.end = now;

                const type = last.type === 'WORK'? 'BREAK' : 'WORK';

                entries.push({ type: type, start: now, end: null });

                updateUI(type === 'BREAK'? 'PAUSED' : 'WORKING');

                save();

            }

        }


        function clockOut() {

            const now = Date.now();

            const last = entries[entries.length - 1];

            if (last.end === null) last.end = now;

            updateUI('IDLE');

            save();

        }


        function updateUI(state) {

            const last = entries[entries.length - 1];

            document.getElementById('startBtn').disabled = (state!== 'IDLE');

            document.getElementById('pauseBtn').disabled = (state === 'IDLE');

            document.getElementById('stopBtn').disabled = (state === 'IDLE');

            document.getElementById('pauseBtn').innerText = (state === 'PAUSED'? 'Weiter' : 'Pause');

            document.getElementById('status').innerText = state;


            if (state === 'IDLE') {

                clearInterval(timerInterval);

                document.getElementById('display').innerText = "00:00:00";

            } else {

                clearInterval(timerInterval);

                timerInterval = setInterval(updateTimer, 1000);

            }

        }


        function updateTimer() {

            const last = entries[entries.length - 1];

            const diff = Date.now() - last.start;

            document.getElementById('display').innerText = new Date(diff).toISOString().substr(11, 8);

        }


        function renderStats() {

            const stats = { day: 0, week: 0, month: 0, year: 0 };

            const now = new Date();


            entries.forEach(e => {

                if (e.type === 'WORK' && e.end) {

                    const d = new Date(e.start);

                    const duration = e.end - e.start;

                    if (d.toDateString() === now.toDateString()) stats.day += duration;

                    if (isSameWeek(d, now)) stats.week += duration;

                    if (d.getMonth() === now.getMonth() && d.getFullYear() === now.getFullYear()) stats.month += duration;

                    if (d.getFullYear() === now.getFullYear()) stats.year += duration;

                }

            });


            document.getElementById('statsContent').innerHTML = `

                <div class="stat-row"><span>Heute:</span> <b>${formatMs(stats.day)}</b></div>

                <div class="stat-row"><span>Woche:</span> <b>${formatMs(stats.week)}</b></div>

                <div class="stat-row"><span>Monat:</span> <b>${formatMs(stats.month)}</b></div>

                <div class="stat-row"><span>Jahr:</span> <b>${formatMs(stats.year)}</b></div>

            `;

        }


        function isSameWeek(d1, d2) {

            const oneJan = new Date(d1.getFullYear(),0,1);

            const w1 = Math.ceil((((d1 - oneJan) / 86400000) + oneJan.getDay() + 1) / 7);

            const w2 = Math.ceil((((d2 - oneJan) / 86400000) + oneJan.getDay() + 1) / 7);

            return w1 === w2 && d1.getFullYear() === d2.getFullYear();

        }


        function formatMs(ms) {

            const h = Math.floor(ms / 3600000);

            const m = Math.floor((ms % 3600000) / 60000);

            return `${h}h ${m}m`;

        }


        function exportCSV() {

            let csv = "Datum;Typ;Start;Ende;Dauer(Min)\n";

            entries.forEach(e => {

                const duration = e.end? Math.round((e.end - e.start) / 60000) : 0;

                csv += `${new Date(e.start).toLocaleDateString()};${e.type};${new Date(e.start).toLocaleTimeString()};${e.end? new Date(e.end).toLocaleTimeString() : '-'};${duration}\n`;

            });

            const blob = new Blob(["\ufeff" + csv], { type: 'text/csv;charset=utf-8;' });

            const url = URL.createObjectURL(blob);

            const link = document.createElement("a");

            link.setAttribute("href", url);

            link.setAttribute("download", `Arbeitszeit_${new Date().toISOString().slice(0,10)}.csv`);

            link.click();

        }


        // Service Worker Registrierung

        if ('serviceWorker' in navigator) {

            navigator.serviceWorker.register('sw.js');

        }


        renderStats();

    </script>
